1. Create a procedure to find greatest of 3 numbers. 

mysql> delimiter //
mysql> create procedure greatest(a int,b int,c int)
    -> begin
    -> declare x,y,z int;
    -> set x=a;
    -> set y=b;
    -> set z=c;
    -> if x>y and x>z then
    -> select x as 'greater';
    -> elseif y>x and y>z then 
    -> select y as 'greater';
    -> else
    -> select z as 'greater';
    -> end if;
    -> end //

2. Create a procedure to display invoice details of a particular customer.

mysql> delimiter //
mysql> create procedure i_d(in cid int) begin select customer2.id,customer2.name,invoice_master.inv_id,invoice_master.inv_date,invoice_item.pid,invoice_item.quantity from customer2 join invoice_master on customer2.id=invoice_master.id join invoice_item on invoice_master.inv_id=invoice_item.inv_id where customer2.id=cid;end//
