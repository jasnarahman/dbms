2. Create a view for invoice products[cname, , inv_id, product_id, quantity]

mysql> create view invoice_product  as select customer2.name,invoice_item.inv_id,invoice_item.pid,invoice_item.quantity from customer2 join invoice_master on customer2.id=invoice_master.id join invoice_item on invoice_master.inv_id=invoice_item.inv_id;
