---------------------------------------------------------------------------------------
mysql> select customer2.name, product2.pname, invoice_master.inv_date from customer2 join invoice_master on customer2.id = invoice_master.id join invoice_item on invoice_master.inv_id = invoice_item.inv_id join product2 on invoice_item.pid = product2.pid where product2.pname = 'eraser';
